TD : Apprentissage fédéré avec FashionMNIST

Objectifs

  • Comprendre les principes de l'apprentissage fédéré (FL) et le rôle du client, du serveur et de l'agrégation.
  • Mettre en place un système FL avec PyTorch et Flower (mode simulation).
  • Entraîner un classifieur d'images (FashionMNIST) en mode fédéré.
  • Comparer les performances du modèle fédéré à celles d'un entraînement centralisé.

Rappel : l'apprentissage fédéré

L'apprentissage fédéré est une approche de machine learning où plusieurs clients (appareils, organisations) entraînent collaborativement un modèle sans partager leurs données brutes : chaque client entraîne localement, puis envoie une mise à jour du modèle qu'un serveur agrège (FedAvg : moyenne pondérée par la taille des jeux de données locaux).

Attention : ne pas partager les données brutes ne garantit pas à lui seul la confidentialité : les mises à jour peuvent elles-mêmes fuiter de l'information (voir questions finales).

2. Modèles

Deux modèles sont utilisés dans les TPs et le projet. Placez-les dans models.py.

import torch
import torch.nn as nn
import torch.nn.functional as F


class LeNet(nn.Module):
    """LeNet-5 adapté à FashionMNIST (entrée 1x28x28)."""
    def __init__(self, num_classes=10):
        super().__init__()
        self.conv1 = nn.Conv2d(1, 6, kernel_size=5, stride=1, padding=2)   # -> 6x28x28
        self.conv2 = nn.Conv2d(6, 16, kernel_size=5, stride=1)             # -> 16x10x10
        self.fc1 = nn.Linear(16 * 5 * 5, 120)
        self.fc2 = nn.Linear(120, 84)
        self.fc3 = nn.Linear(84, num_classes)

    def forward(self, x):
        x = F.max_pool2d(F.relu(self.conv1(x)), 2, 2)                     # -> 6x14x14
        x = F.max_pool2d(F.relu(self.conv2(x)), 2, 2)                     # -> 16x5x5
        x = torch.flatten(x, start_dim=1)
        x = F.relu(self.fc1(x))
        x = F.relu(self.fc2(x))
        return self.fc3(x)


class CNN(nn.Module):
    """CNN simple (entrée 1x28x28)."""
    def __init__(self, num_classes=10):
        super().__init__()
        self.conv1 = nn.Conv2d(1, 8, kernel_size=3, padding=1)             # -> 8x28x28
        self.conv2 = nn.Conv2d(8, 16, kernel_size=3, padding=1)            # -> 16x14x14
        self.fc1 = nn.Linear(16 * 7 * 7, 64)
        self.fc2 = nn.Linear(64, num_classes)

    def forward(self, x):
        x = F.max_pool2d(F.relu(self.conv1(x)), 2, 2)                     # -> 8x14x14
        x = F.max_pool2d(F.relu(self.conv2(x)), 2, 2)                     # -> 16x7x7
        x = torch.flatten(x, start_dim=1)
        x = F.relu(self.fc1(x))
        return self.fc2(x)

_fds = None  # jeu de données fédéré, créé une seule fois par processus


def load_data(partition_id: int, num_partitions: int, batch_size: int):
    """Charge la partition `partition_id` (IID) et la découpe en train/validation (80/20)."""
    global _fds
    if _fds is None:
        partitioner = IidPartitioner(num_partitions=num_partitions)
        _fds = FederatedDataset(dataset="zalando-datasets/fashion_mnist",
                                partitioners={"train": partitioner})
    partition = _fds.load_partition(partition_id)
    split = partition.train_test_split(test_size=0.2, seed=42)
    tf = Compose([ToTensor(), Normalize((0.2860,), (0.3530,))])

    def apply_tf(batch):
        batch["image"] = [tf(img) for img in batch["image"]]
        return batch

    split = split.with_transform(apply_tf)
    return (DataLoader(split["train"], batch_size=batch_size, shuffle=True),
            DataLoader(split["test"], batch_size=batch_size))

4. Client et serveur

client_app.py :

import torch
from flwr.client import ClientApp, NumPyClient
from flwr.common import Context

from fl_on_fashionmnist.models import LeNet
from fl_on_fashionmnist.task import get_weights, set_weights, load_data, train, test


class FlowerClient(NumPyClient):
    def __init__(self, net, trainloader, valloader, local_epochs):
        self.net, self.trainloader, self.valloader = net, trainloader, valloader
        self.local_epochs = local_epochs
        self.device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
        self.net.to(self.device)

    def fit(self, parameters, config):
        set_weights(self.net, parameters)
        loss = train(self.net, self.trainloader, self.local_epochs, self.device)
        return get_weights(self.net), len(self.trainloader.dataset), {"train_loss": loss}

    def evaluate(self, parameters, config):
        set_weights(self.net, parameters)
        loss, acc = test(self.net, self.valloader, self.device)
        return loss, len(self.valloader.dataset), {"accuracy": acc}


def client_fn(context: Context):
    partition_id = context.node_config["partition-id"]
    num_partitions = context.node_config["num-partitions"]
    local_epochs = context.run_config["local-epochs"]
    batch_size = context.run_config["batch-size"]
    trainloader, valloader = load_data(partition_id, num_partitions, batch_size)
    return FlowerClient(LeNet(), trainloader, valloader, local_epochs).to_client()


app = ClientApp(client_fn=client_fn)

server_app.py :

from flwr.common import Context, ndarrays_to_parameters
from flwr.server import ServerApp, ServerAppComponents, ServerConfig
from flwr.server.strategy import FedAvg

from fl_on_fashionmnist.models import LeNet
from fl_on_fashionmnist.task import get_weights


def weighted_average(metrics):
    """Moyenne pondérée par le nombre d'exemples des métriques renvoyées par les clients."""
    total = sum(n for n, _ in metrics)
    return {"accuracy": sum(n * m["accuracy"] for n, m in metrics) / total}


def server_fn(context: Context):
    num_rounds = context.run_config["num-server-rounds"]
    fraction_fit = context.run_config["fraction-fit"]
    initial = ndarrays_to_parameters(get_weights(LeNet()))
    strategy = FedAvg(fraction_fit=fraction_fit, fraction_evaluate=1.0,
                      min_available_clients=2, initial_parameters=initial,
                      evaluate_metrics_aggregation_fn=weighted_average)
    return ServerAppComponents(strategy=strategy, config=ServerConfig(num_rounds=num_rounds))


app = ServerApp(server_fn=server_fn)

5. Configuration et exécution

pyproject.toml (exemple à adapter selon votre version de Flower) :

[build-system]
requires = ["hatchling"]
build-backend = "hatchling.build"

[project]
name = "fl_on_fashionmnist"
version = "1.0.0"
description = "FL sur FashionMNIST avec Flower"
dependencies = [
    "flwr[simulation]>=1.13.0",
    "flwr-datasets[vision]>=0.5.0",
    "torch>=2.2.0",
    "torchvision>=0.17.0",
]

[tool.hatch.build.targets.wheel]
packages = ["."]

[tool.flwr.app]
publisher = "votre-nom"

[tool.flwr.app.components]
serverapp = "fl_on_fashionmnist.server_app:app"
clientapp = "fl_on_fashionmnist.client_app:app"

[tool.flwr.app.config]
num-server-rounds = 3
fraction-fit = 0.5
local-epochs = 1
batch-size = 32
learning-rate = 0.01

[tool.flwr.federations]
default = "local-simulation"

[tool.flwr.federations.local-simulation]
options.num-supernodes = 10
flwr run .
Remarque : les anciennes versions de ce TD utilisaient fl.server.start_server(...) et python server.py / python client.py. Cette API est dépréciée dans les versions récentes de Flower ; le mode simulation avec flwr run ci-dessus la remplace.
